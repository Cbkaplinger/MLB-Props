def _write(tmp: Path, name: str, *, row_group_size: int = 6,
           write_statistics: bool = True, interleave: bool = False,
           n_pairs: int = 1) -> Path:
    """Write a synthetic book_lines parquet. ``interleave`` orders rows
    so consecutive rows alternate 2025/2026 events (mixed row groups);
    otherwise rows are grouped by event (separable layout)."""
    rows = _rows(SPEC_ORDERED)
    if interleave:
        base = list(rows)
        a, b = base[: len(base) // 2], base[len(base) // 2:]
        rows = [r for pair in zip(a, b) for r in pair]
    path = tmp / name
    pq.write_table(pa.Table.from_pylist(rows), path,
                   row_group_size=row_group_size,
                   write_statistics=write_statistics)
    return path
